cmake_minimum_required(VERSION 3.23)
project(image_dependency_smoke LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(variable DEPS_ROOT TYPE PLATFORM)
    if(NOT DEFINED ${variable})
        message(FATAL_ERROR "Pass -D${variable}=...")
    endif()
endforeach()
set(suffix "lib/${TYPE}/${PLATFORM}")
# Require the staged exports, never a Homebrew/distro copy of these libraries.
set(Imath_DIR "${DEPS_ROOT}/imath/${suffix}/cmake/Imath")
set(libjpeg-turbo_DIR "${DEPS_ROOT}/libjpeg-turbo/${suffix}/cmake/libjpeg-turbo")
if(TYPE STREQUAL "vs")
    set(ZLIB_LIBRARY "${DEPS_ROOT}/zlib/${suffix}/zlib.lib" CACHE FILEPATH "" FORCE)
else()
    set(ZLIB_LIBRARY "${DEPS_ROOT}/zlib/${suffix}/zlib.a" CACHE FILEPATH "" FORCE)
endif()
set(ZLIB_INCLUDE_DIR "${DEPS_ROOT}/zlib/include" CACHE PATH "" FORCE)
find_package(Imath CONFIG REQUIRED PATHS "${Imath_DIR}" NO_DEFAULT_PATH)
find_package(libjpeg-turbo CONFIG REQUIRED PATHS "${libjpeg-turbo_DIR}" NO_DEFAULT_PATH)
find_package(OpenEXR CONFIG REQUIRED PATHS "${DEPS_ROOT}/openexr/${suffix}/cmake/OpenEXR" NO_DEFAULT_PATH)
find_package(Tiff CONFIG REQUIRED PATHS "${DEPS_ROOT}/libtiff/${suffix}/cmake/tiff" NO_DEFAULT_PATH)
if(NOT EXISTS "${TIFF_INCLUDE_DIR}/tiffio.h")
    message(FATAL_ERROR "TIFF_INCLUDE_DIR is not relocatable: ${TIFF_INCLUDE_DIR}")
endif()
add_executable(image-dependency-smoke smoke.cpp)
target_link_libraries(image-dependency-smoke PRIVATE
    Imath::Imath OpenEXR::OpenEXR TIFF::tiff
    libjpeg-turbo::jpeg libjpeg-turbo::turbojpeg)
enable_testing()
add_test(NAME image-dependency-smoke COMMAND image-dependency-smoke)

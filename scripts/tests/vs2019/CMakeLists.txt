cmake_minimum_required(VERSION 3.21)
project(vs2019_maximum LANGUAGES CXX)
if(NOT MSVC OR MSVC_VERSION LESS 1920 OR MSVC_VERSION GREATER_EQUAL 1930)
    message(FATAL_ERROR "Expected the VS2019 compiler, got ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
endif()
if(NOT CMAKE_VS_PLATFORM_TOOLSET STREQUAL "v142")
    message(FATAL_ERROR "Expected v142, got ${CMAKE_VS_PLATFORM_TOOLSET}")
endif()
add_executable(vs2019_maximum main.cpp)
# On MSVC 19.29, this selects /std:c++latest, rather than /std:c++20.
set_target_properties(vs2019_maximum PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED YES)
if(SDK_ROOT)
    if(BUNDLE STREQUAL "1")
        target_include_directories(vs2019_maximum PRIVATE "${SDK_ROOT}/zlib/include")
        target_link_libraries(vs2019_maximum PRIVATE "${SDK_ROOT}/zlib/lib/vs/x64/zlib.lib")
        target_compile_definitions(vs2019_maximum PRIVATE TEST_PACKAGED_ZLIB)
    elseif(BUNDLE STREQUAL "2")
        target_include_directories(vs2019_maximum PRIVATE "${SDK_ROOT}/fmt/include")
        target_link_libraries(vs2019_maximum PRIVATE "${SDK_ROOT}/fmt/lib/vs/x64/fmt.lib")
        target_compile_definitions(vs2019_maximum PRIVATE TEST_PACKAGED_FMT)
    else()
        message(FATAL_ERROR "Expected bundle 1 or 2")
    endif()
endif()

cmake_minimum_required(VERSION 3.16)
project(ProtobufPackageConsumer LANGUAGES CXX)
find_package(protobuf CONFIG REQUIRED)
enable_testing()
foreach(runtime IN ITEMS full lite)
    add_executable(${runtime} smoke.cpp "${GENERATED_DIR}/${runtime}.pb.cc")
    target_compile_features(${runtime} PRIVATE cxx_std_17)
    target_include_directories(${runtime} PRIVATE "${GENERATED_DIR}")
    if(runtime STREQUAL "lite")
        target_compile_definitions(${runtime} PRIVATE TEST_LITE)
        target_link_libraries(${runtime} PRIVATE protobuf::libprotobuf-lite)
    else()
        target_link_libraries(${runtime} PRIVATE protobuf::libprotobuf)
    endif()
    add_test(NAME ${runtime}-roundtrip COMMAND ${runtime})
endforeach()

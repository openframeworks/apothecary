cmake_minimum_required(VERSION 3.23)
project(oqs_smoke LANGUAGES C)
set(CMAKE_C_STANDARD 11)
foreach(variable DEPS_ROOT TYPE PLATFORM)
    if(NOT DEFINED ${variable})
        message(FATAL_ERROR "Pass -D${variable}=...")
    endif()
endforeach()
set(libdir "lib/${TYPE}/${PLATFORM}")
find_package(liboqs CONFIG REQUIRED PATHS "${DEPS_ROOT}/liboqs/${libdir}/cmake/liboqs" NO_DEFAULT_PATH)
find_package(Threads REQUIRED)
set(OPENSSL_INCLUDE_DIR "${DEPS_ROOT}/openssl/include" CACHE PATH "" FORCE)
set(OPENSSL_CRYPTO_LIBRARY "${DEPS_ROOT}/openssl/${libdir}/libcrypto.a" CACHE FILEPATH "" FORCE)
set(OPENSSL_SSL_LIBRARY "${DEPS_ROOT}/openssl/${libdir}/libssl.a" CACHE FILEPATH "" FORCE)
set(OPENSSL_USE_STATIC_LIBS ON)
set(ZLIB_LIBRARY "${DEPS_ROOT}/zlib/${libdir}/zlib.a" CACHE FILEPATH "" FORCE)
set(ZLIB_INCLUDE_DIR "${DEPS_ROOT}/zlib/include" CACHE PATH "" FORCE)
find_package(oqsprovider CONFIG REQUIRED PATHS "${DEPS_ROOT}/oqsprovider/${libdir}/cmake" NO_DEFAULT_PATH)
set(openssl_libraries
    "${DEPS_ROOT}/openssl/${libdir}/libssl.a"
    "${DEPS_ROOT}/openssl/${libdir}/libcrypto.a"
    "${DEPS_ROOT}/zlib/${libdir}/zlib.a"
    Threads::Threads ${CMAKE_DL_LIBS})
foreach(variant module static)
    add_executable(oqs-${variant}-smoke provider.c)
    target_include_directories(oqs-${variant}-smoke PRIVATE "${DEPS_ROOT}/openssl/include")
    if(variant STREQUAL "static")
        target_compile_definitions(oqs-${variant}-smoke PRIVATE STATIC_OQS_PROVIDER)
        target_link_libraries(oqs-${variant}-smoke PRIVATE
            apothecary::oqsprovider)
    endif()
    target_link_libraries(oqs-${variant}-smoke PRIVATE ${openssl_libraries})
endforeach()
add_executable(liboqs-smoke liboqs.c)
target_link_libraries(liboqs-smoke PRIVATE OQS::oqs)
enable_testing()
add_test(NAME liboqs-smoke COMMAND liboqs-smoke)
add_test(NAME oqs-static-smoke COMMAND oqs-static-smoke)
add_test(NAME oqs-module-smoke COMMAND oqs-module-smoke "${DEPS_ROOT}/oqsprovider/${libdir}/ossl-modules")

cmake_minimum_required(VERSION 3.22)
project(rgfw VERSION 1.8.1 LANGUAGES C)

add_library(rgfw STATIC rgfw.c)
target_compile_features(rgfw PUBLIC c_std_99)
target_compile_definitions(rgfw PUBLIC RGFW_OPENGL RGFWDEF=extern)
target_include_directories(rgfw PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>" "$<INSTALL_INTERFACE:include>")
set_target_properties(rgfw PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(EMSCRIPTEN)
    # Threading/memory64 are supplied by Apothecary's selected build profile.
elseif(WIN32)
    target_link_libraries(rgfw PUBLIC opengl32 gdi32 user32 shell32)
elseif(APPLE)
    target_link_libraries(rgfw PUBLIC "-framework Cocoa" "-framework OpenGL" "-framework IOKit" "-framework CoreVideo")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(X11 REQUIRED COMPONENTS Xrandr Xcursor Xi Xext)
    find_package(OpenGL REQUIRED)
    find_package(Threads REQUIRED)
    target_link_libraries(rgfw PUBLIC X11::X11 X11::Xrandr X11::Xcursor X11::Xi X11::Xext OpenGL::GL Threads::Threads ${CMAKE_DL_LIBS} m)
else()
    message(FATAL_ERROR "RGFW 1.8.1 has no backend for ${CMAKE_SYSTEM_NAME}")
endif()
install(TARGETS rgfw ARCHIVE DESTINATION lib)
install(FILES RGFW.h XDL.h rgfw_apothecary.h DESTINATION include)
install(FILES LICENSE DESTINATION license)
